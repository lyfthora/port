Go

Go is an open-source programming language developed by Google in 2009. It was designed to be simple, efficient, and easy to maintain for large-scale projects. It compiles directly to native machine code, producing fast and lightweight executables that do not require external runtimes. Its type system is static, but includes type inference, reducing verbosity without sacrificing compile-time safety. One of its most distinctive features is its concurrency model: using goroutines and channels, it allows running thousands of tasks in parallel with minimal memory usage, without manually managing threads. This makes it especially suitable for servers, APIs, and systems handling many simultaneous connections. 

router := gin.Default()
router.Use(middleware.Cors())

// rate limiting
limiter := middleware.NewLimiter(rate.Every(2*time.Second), 5)
router.Use(middleware.RateLimit(limiter))

router.GET("/health", func(c *gin.Context) {
	c.JSON(http.StatusOK, gin.H{"status": "ok"})
})

userController := controllers.NewUserController()
projectController := controllers.NewProjectController()
authController := controllers.NewAuthController()

api := router.Group("api")
{
	// Auth
	api.POST("/auth/login", authController.Login)

	// users
	api.POST("/user", userController.CreateUser)

	// Protected routes
	authRequired := api.Group("/")
	authRequired.Use(middleware.Authenticate())
	{
		authRequired.POST("/userData", userController.GetUserData)

		// projects
		authRequired.GET("/projects", projectController.GetProjects)
		authRequired.POST("/projects", projectController.CreateProject)

		// Project-specific routes
		projectSpecific := authRequired.Group("/projects/:projectId")
		projectSpecific.Use(middleware.OwnershipAuth())
		{
			projectSpecific.PATCH("", projectController.PatchProject)
			projectSpecific.PUT("/duplicate", projectController.DuplicateProject)
			projectSpecific.DELETE("", projectController.DeleteProject)
		}
	}
}
port := cfg.Port
log.Printf("Server running on port %s", port)
log.Fatal(router.Run(":" + port))
